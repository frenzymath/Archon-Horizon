\chapter{Connectedness}

\begin{theorem}[Finite discrete connectedness]
\label{thm:discrete-connected}
\lean{MiniTopology.discrete_connected_iff}
\notready
A finite discrete space is connected exactly when it has at most one point.
\uses{lem:singleton-open}
\end{theorem}

\begin{remark}[The next bridge]
\label{rem:clopen-bridge}
The missing proof should turn a nontrivial finite partition into a clopen
separation, then use the connectedness definition.
\end{remark}
