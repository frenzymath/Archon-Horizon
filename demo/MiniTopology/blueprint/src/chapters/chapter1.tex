\chapter{Finite Spaces}

\begin{definition}[Finite compactness]
\label{def:finite-compact}
\lean{MiniTopology.finite_compact}
\leanok
A finite subset of the natural numbers is compact in the demo topology.
\end{definition}

\begin{lemma}[Singleton opens]
\label{lem:singleton-open}
\lean{MiniTopology.singleton_is_open}
Every singleton is open in the discrete demo space.
\uses{def:finite-compact}
\leanok
\end{lemma}
